\section*{Algorithm Pseudocode}
\label{app:pseudocode}

\subsection*{Single-Objective Genetic Algorithm}

\begin{algorithm}[H]
\caption{Single-Objective Genetic Algorithm for Game Balancing}
\label{alg:ga}
\begin{algorithmic}[1]
\Require Population size $N=100$, generations $G=30$, simulations per genome $S=200$
\State $P \gets \text{InitPopulation}(N)$  \Comment{Each genome $\sim$ $\mathcal{U}(0.7, 1.3)^{500}$}
\For{$g = 1$ \textbf{to} $G$}
    \State $\text{EvaluateParallel}(P, S)$  \Comment{8-core parallel eval.}
    \State $P_{\text{elite}} \gets \text{TopK}(P, 0.1N)$  \Comment{Elitism: keep top 10\%}
    \State $P_{\text{offspring}} \gets \emptyset$
    \While{$|P_{\text{offspring}}| < N - |P_{\text{elite}}|$}
        \State $p_1 \gets \text{TournamentSelect}(P, k=3)$
        \State $p_2 \gets \text{TournamentSelect}(P, k=3)$
        \State $c \gets \text{UniformCrossover}(p_1, p_2, p_c=0.80)$
        \State $c \gets \text{GaussianMutate}(c, p_m=0.02, \sigma=0.10)$
        \State $P_{\text{offspring}} \gets P_{\text{offspring}} \cup \{c\}$
    \EndWhile
    \State $P \gets P_{\text{elite}} \cup P_{\text{offspring}}$
\EndFor
\State \Return $\text{BestFitness}(P)$
\end{algorithmic}
\end{algorithm}

\subsection*{NSGA-II Multi-Objective Optimizer}

\begin{algorithm}[H]
\caption{NSGA-II for Multi-Objective Game Balancing}
\label{alg:nsga2}
\begin{algorithmic}[1]
\Require Population size $N=50$, generations $G=30$
\State $P \gets \text{InitPopulation}(N)$
\State $\text{Evaluate}(P, S=80)$  \Comment{20 simulations $\times$ 4 agents per genome}
\For{$g = 1$ \textbf{to} $G$}
    \State $\mathcal{F} \gets \text{FastNonDominatedSort}(P)$  \Comment{Returns fronts $F_1, F_2, \ldots$}
    \State $\text{ComputeCrowdingDistance}(\mathcal{F})$
    \State $Q \gets \text{BinaryTournamentSelect}(P)$  \Comment{By rank then crowding dist.}
    \State $Q \gets \text{SBXCrossover}(Q, \eta_c=20, p_c=0.90)$
    \State $Q \gets \text{PolynomialMutate}(Q, \eta_m=20, p_m=0.05)$
    \State $R \gets P \cup Q$  \Comment{Combined parent + offspring}
    \State $\text{Evaluate}(R \setminus P, S=80)$  \Comment{Evaluate new offspring, 20 sims $\times$ 4 agents}
    \State $\mathcal{F}' \gets \text{FastNonDominatedSort}(R)$
    \State $\text{ComputeCrowdingDistance}(\mathcal{F}')$
    \State $P \gets \text{SelectBestN}(R, N)$  \Comment{By rank, then crowding distance}
\EndFor
\State \Return $F_1$ (first Pareto front of final $P$)
\end{algorithmic}
\end{algorithm}

\subsection*{Heuristic AI Decision Logic}
\begin{lstlisting}[language=Python, caption={Combat decision logic of HeuristicPlayerAI (simplified)}, label=lst:heuristic_ai]
class HeuristicPlayerAI(IPlayerAgent):

    def get_combat_decision(self, run) -> CombatDecision:
        hero    = run.the_hero
        enemies = [e for e in run.current_combat.enemies
                   if e.current_health > 0]
        incoming = self._calc_incoming_damage(run, enemies)
        needed_block = max(0, incoming - hero.block)

        # Priority 1: Survival
        if needed_block > 0:
            move = self._find_best_defensive_move(hero, needed_block)
            if move: return move

        # Priority 2: Lethal
        move = self._find_lethal_move(hero, enemies)
        if move: return move

        # Priority 3: Best general offensive move
        return self._get_best_general_move(hero, enemies)

    def choose_map_node(self, run) -> int:
        hp_pct = run.the_hero.current_health / run.the_hero.max_health
        return max(run.the_map.get_possible_next_nodes(),
                   key=lambda n: self._evaluate_room(run, n, hp_pct)).id
\end{lstlisting}

OBS_DIM = 152   # see Table 3.2
ACT_DIM = 67    # see Section 3.3.2

class RoguelikeEnv(gym.Env):
    def reset(self, *, seed=None, options=None):
        if self._domain_randomization:
            self._apply_domain_randomization()
        self._controller = GameController(
            self._card_pool, self._relic_pool, ...)
        self._controller.start_new_run(seed, self._hero_data)
        return self._get_obs(), {}

    def step(self, action: int):
        run  = self._controller.current_run
        self._apply_action(action, run)
        obs    = self._get_obs()
        reward = self._compute_reward(run)
        done   = (run.current_state == GameState.GameOver)
        info   = {"win": run.is_victory, "floor": run.current_floor}
        return obs, reward, done, False, info

    def action_masks(self) -> np.ndarray:
        mask = np.zeros(ACT_DIM, dtype=bool)
        for a in self._get_valid_actions():
            mask[a] = True
        return mask

\subsection*{Curriculum Domain Randomization}
\begin{lstlisting}[language=Python, caption={Curriculum Domain Randomization in reset()}, label=lst:domain_rand]
def _apply_domain_randomization(self) -> None:
    rng = self.np_random
    if self._dr_curriculum:
        # Ramp noise linearly from 0 to dr_noise over dr_warmup_episodes
        progress = min(self._episode_count /
                       max(self._dr_warmup_episodes, 1), 1.0)
        noise = self._dr_noise * progress
    else:
        noise = self._dr_noise
    lo, hi = 1.0 - noise, 1.0 + noise

    for cid, card in self._card_pool.cards_by_id.items():
        orig_mana, orig_vals = self._orig_card_values[cid]
        card.mana_cost = max(1, round(orig_mana * rng.uniform(lo, hi)))
        for i, action in enumerate(card.actions):
            if orig_vals[i] > 0:
                action.value = max(1, round(orig_vals[i] *
                                            rng.uniform(lo, hi)))

    for eid, enemy in self._enemy_pool.enemies_by_id.items():
        enemy.starting_health = max(
            1, round(self._orig_enemy_hp[eid] * rng.uniform(lo, hi)))

    self._hero_data.starting_health = max(
        1, round(self._orig_hero_hp * rng.uniform(lo, hi)))
\end{lstlisting}

\subsection*{Batch Genome Evaluation}

class SimulationRunner:
    def run_batch(self, genome: HierarchicalGenome,
                  n_runs: int = 30,
                  base_seed: int = 42) -> List[SimulationStats]:
        # Clone base data pools and apply genome
        card_pool  = copy.deepcopy(self._base_card_pool)
        enemy_pool = copy.deepcopy(self._base_enemy_pool)
        hero       = copy.deepcopy(self._base_hero)
        hierarchical_applicator.apply(genome, card_pool, enemy_pool, hero)

        results = []
        for i in range(n_runs):
            agent = self.agents[i % len(self.agents)]  # round-robin
            stats = self._run_single(agent, card_pool, enemy_pool,
                                     hero, seed=base_seed + i)
            results.append(stats)
        return results
